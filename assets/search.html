from pathlib import Path
from contextlib import contextmanager
import base64
import streamlit as st

from theme import inject_css  # si ton fichier s'appelle theme.py à la racine


APP_NAME = "INSIGHTIA"
TAGLINE = "Transformer la voix du client en décisions actionnables"

BASE_DIR = Path(__file__).resolve().parent
ASSETS = BASE_DIR / "assets"
LOGO_PATH = ASSETS / "insightia_logo.svg"

PAGES = [
    ("home", "Accueil"),
    ("solution", "Solution"),
    ("use_cases", "Cas d’usage"),
    ("method", "Méthode"),
    ("contact", "Contact"),
    ("app_import", "Application"),
]


def _img_b64(path: Path) -> str | None:
    if not path.exists():
        return None
    return base64.b64encode(path.read_bytes()).decode("utf-8")


def get_page_key() -> str:
    key = st.query_params.get("page", "home")
    valid = {k for k, _ in PAGES}
    return key if key in valid else "home"


def set_page(key: str):
    st.query_params["page"] = key
    st.rerun()


def render_topnav(current_key: str):
    logo_b64 = _img_b64(LOGO_PATH)

    st.markdown("<div class='topnav'>", unsafe_allow_html=True)

    left, mid, right = st.columns([1.2, 3.8, 1.6], vertical_alignment="center")

    with left:
        if logo_b64:
            st.markdown(
                f"""
                <div style="display:flex;align-items:center;gap:10px;">
                  <img src="data:image/png;base64,{logo_b64}" style="height:30px;border-radius:10px;"/>
                  <div class="brand">{APP_NAME}</div>
                </div>
                """,
                unsafe_allow_html=True,
            )
        else:
            st.markdown(f"<div class='brand'>{APP_NAME}</div>", unsafe_allow_html=True)

    with mid:
        c = st.columns(5)
        nav_keys = ["home", "solution", "use_cases", "method", "contact"]
        labels = ["Accueil", "Solution", "Cas d’usage", "Méthode", "Contact"]
        for i, (k, label) in enumerate(zip(nav_keys, labels)):
            with c[i]:
                disabled = (k == current_key)
                if st.button(label, key=f"nav_{k}", disabled=disabled, use_container_width=True):
                    set_page(k)

    with right:
        # CTA style "Essayez Gratuitement"
        if st.button("Essayez Gratuitement", type="primary", use_container_width=True):
            set_page("app_import")

    st.markdown("</div>", unsafe_allow_html=True)
